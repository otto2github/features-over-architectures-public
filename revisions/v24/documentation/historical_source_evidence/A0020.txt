L00219 | def derive_company_company_edges(company_holder_df: pd.DataFrame,
L00220 |                                   edge_type: str,
L00221 |                                   attr_name: str,
L00222 |                                   min_shared: int,
L00223 |                                   logger) -> pd.DataFrame:
L00224 |     """
L00225 |     通过 (company, holder, year) 派生 (company_a, company_b, year, shared_holder, weight).
L00226 |     """
L00227 |     if company_holder_df is None or company_holder_df.empty:
L00228 |         return pd.DataFrame()
L00229 |     df = company_holder_df[["src", "dst", "year"]].copy()
L00230 |     df.columns = ["company", "holder", "year"]
L00231 | 
L00232 |     # 同 (holder, year) groupby, 笛卡尔积公司 → 公司对
L00233 |     pairs = []
L00234 |     grp = df.groupby(["holder", "year"])
L00235 |     n_groups = len(grp)
L00236 |     logger.info("[%s] 共 %d 个 (holder, year) 组待派生", edge_type, n_groups)
L00237 |     cnt = 0
L00238 |     skipped_huge = 0
L00239 |     for (holder, year), g in grp:
L00240 |         cs = g["company"].unique()
L00241 |         if len(cs) < 2:
L00242 |             continue
L00243 |         # 大型 holder (e.g. 香港中央结算) 持有数百上市公司, 笛卡尔积爆炸 → 跳过
L00244 |         if len(cs) > 50:
L00245 |             skipped_huge += 1
L00246 |             continue
L00247 |         cs_sorted = sorted(cs)
L00248 |         for i, ca in enumerate(cs_sorted):
L00249 |             for cb in cs_sorted[i + 1:]:
L00250 |                 pairs.append((ca, cb, year, holder))
L00251 |         cnt += 1
L00252 | 
L00253 |     logger.info("[%s] 派生 %d 个公司对, 跳过 %d 个大型 holder (>50 公司)",
L00254 |                 edge_type, len(pairs), skipped_huge)
L00255 | 
L00256 |     if not pairs:
L00257 |         return pd.DataFrame()
L00258 | 
L00259 |     out = pd.DataFrame(pairs, columns=["src", "dst", "year", attr_name])
L00260 |     # weight = 同 (src, dst, year) 共享 holder 数
L00261 |     weight = (out.groupby(["src", "dst", "year"])
L00262 |                  .size().reset_index(name="weight"))
L00263 |     out_agg = (out.groupby(["src", "dst", "year"])
L00264 |                   [attr_name].apply(lambda s: ",".join(map(str, sorted(set(s)))))
L00265 |                   .reset_index())
L00266 |     out_agg = out_agg.merge(weight, on=["src", "dst", "year"])
L00267 |     out_agg["edge_type"] = edge_type
L00268 |     out_agg = out_agg[out_agg["weight"] >= min_shared]
L00269 |     logger.info("[%s] 聚合后公司对 %d 条 (min_shared=%d)",
L00270 |                 edge_type, len(out_agg), min_shared)
L00271 |     return out_agg
