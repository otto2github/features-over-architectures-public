L00254 | def export_review_queue(nodes_df: pd.DataFrame, logger) -> int:
L00255 |     """
L00256 |     导出可疑节点供人工复核:
L00257 |       - canonical_name 长度 ≤ 1 (异常短)
L00258 |       - canonical_name 长度 > 30 (异常长)
L00259 |       - n_records == 1 且类型为 institution (一次性出现, 可能是数据错误)
L00260 |       - 自然人但 raw_name_first 含数字 / ASCII (可能误判)
L00261 |     """
L00262 |     issues = []
L00263 |     for _, r in nodes_df.iterrows():
L00264 |         if r["type"] == "company":
L00265 |             continue
L00266 |         cn = r["canonical_name"]
L00267 |         rn = r["raw_name_first"]
L00268 |         n = r["n_records"]
L00269 |         flags = []
L00270 |         if len(cn) <= 1:
L00271 |             flags.append("too_short")
L00272 |         if len(cn) > 30:
L00273 |             flags.append("too_long")
L00274 |         if r["type"] == "institution" and n == 1:
L00275 |             flags.append("singleton_inst")
L00276 |         if r["type"] == "person":
L00277 |             import re as _re
L00278 |             if _re.search(r"[A-Za-z0-9]", rn or ""):
L00279 |                 flags.append("person_has_ascii")
L00280 |             if len(cn) > 6:
L00281 |                 flags.append("person_too_long")
L00282 |         if flags:
L00283 |             issues.append({
L00284 |                 "id": r["id"],
L00285 |                 "type": r["type"],
L00286 |                 "canonical_name": cn,
L00287 |                 "raw_name_first": rn,
L00288 |                 "n_records": n,
L00289 |                 "flags": ",".join(flags),
L00290 |             })
L00291 |     if not issues:
L00292 |         logger.info("审核队列: 无可疑节点")
L00293 |         return 0
L00294 |     df = pd.DataFrame(issues)
L00295 |     out = STATS_DIR / "build_kg_nodes_review_queue.csv"
L00296 |     df.to_csv(out, index=False, encoding="utf-8-sig")
L00297 |     logger.info("审核队列: %d 条, 写入 %s", len(df), out)
L00298 |     return len(df)
