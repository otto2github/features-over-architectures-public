"""
P4.5.1 Step 2: compute_fina_indicator_v0.py
============================================

目标:
  从 fina_indicator 表抽取 51 维 t0 + 12 维 t1 + 15 维 zind 特征,
  align 到 v0 清洁全集 (51,675 firm-years),
  输出 fina_indicator_v0.parquet (51,675 × 81 列)
    = 4 元数据 + 51 t0 + 12 t1 + 15 zind - 1 industry (留作 zind 输入)

决策依据 (I 系列):
  I7   : end_date LIKE '%1231' (仅年报)
  I9.1 : 同 (ts_code, end_date) 取 ann_date 最大者 (替代 update_flag)
  I10  : pd.to_numeric(errors='coerce')
  I11  : firm_id ↔ ts_code 映射从 base 矩阵反查
  I12  : zind 分组 (year, industry), min_samples=10
  I13  : 51 t0 / 12 t1 / 15 zind (本脚本主要决策)
  I13.3.1 : 不过滤 comp_type (fina_indicator 默认合并口径)
  I13.3.2 : ann_date 最大者去重 (无 update_flag 字段)
  I13.3.3 : end_date LIKE '%1231'
  I13.3.4 : 全字段 to_numeric coerce
  I13.3.5 : t1 缺失保 NaN, 不填充
  I13.3.6 : zind (year, industry), min_samples=10
  I13.3.7 : industry NaN → zind 列 NaN
  I13.3.8 : 不做 winsorize

作者: Otto (CYQ)
日期: 2026-05-06
"""

import sqlite3
import logging
import time
from pathlib import Path

import numpy as np
import pandas as pd

# ---------------- 配置 ----------------
PROJECT_ROOT = Path.home() / "code" / "thesis_project"
DB_FIN = PROJECT_ROOT / "data" / "raw" / "financials" / "financials.db"
DB_BASIC = PROJECT_ROOT / "data" / "raw" / "stock_basic.db"

# base: 用 derived_financials_v0 提供 (firm_id, year, ts_code) 映射
BASE_PARQUET = PROJECT_ROOT / "data" / "processed" / "features" / "derived_financials_v0.parquet"

OUTPUT_DIR = PROJECT_ROOT / "data" / "processed" / "features"
OUTPUT_PARQUET = OUTPUT_DIR / "fina_indicator_v0.parquet"

LOG_DIR = PROJECT_ROOT / "data" / "interim" / "p4_5_1_compute"
LOG_PATH = LOG_DIR / "compute_fina_indicator.log"

OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
LOG_DIR.mkdir(parents=True, exist_ok=True)

# ---------------- 字段清单 (I13) ----------------

# 51 维 t0 字段
T0_FIELDS = [
    # LIQUIDITY (3)
    "current_ratio", "quick_ratio", "cash_ratio",
    # TURNOVER (4)
    "assets_turn", "fa_turn", "ca_turn", "turn_days",
    # GROWTH yoy (10) - 已去共线性
    "assets_yoy", "equity_yoy", "bps_yoy", "tr_yoy", "op_yoy",
    "ebt_yoy", "netprofit_yoy", "dt_eps_yoy", "ocf_yoy", "roe_yoy",
    # CASHFLOW (7)
    "ocfps", "cfps", "ocf_to_debt", "ocf_to_shortdebt",
    "q_ocf_to_sales", "fcfe_ps", "fcff_ps",
    # PROFITABILITY 利润率 (7)
    "profit_to_gr", "profit_to_op", "op_of_gr", "gc_of_gr",
    "ebit_of_gr", "roic", "ebit_ps",
    # 费用结构 (4)
    "adminexp_of_gr", "finaexp_of_gr", "expense_of_sales", "saleexp_to_gr",
    # ROA/ROE (3)
    "roa", "roe", "npta",
    # 资本结构 (5)
    "eqt_to_debt", "assets_to_eqt", "debt_to_eqt",
    "eqt_to_talcapital", "eqt_to_interestdebt",
    # 负债/资产结构 (4)
    "currentdebt_to_debt", "tangibleasset_to_debt",
    "ca_to_assets", "tbassets_to_totalassets",
    # PER_SHARE (4)
    "bps", "eps", "undist_profit_ps", "total_revenue_ps",
]
assert len(T0_FIELDS) == 51, f"T0 应 51 个, 实际 {len(T0_FIELDS)}"

# 12 维 t1 字段 (核心比率, 不含 yoy/_ps/占比)
T1_FIELDS = [
    "current_ratio", "quick_ratio",
    "assets_turn",
    "roa", "roe", "npta",
    "profit_to_gr", "op_of_gr",
    "ocf_to_debt", "q_ocf_to_sales",
    "debt_to_eqt", "assets_to_eqt",
]
assert len(T1_FIELDS) == 12

# 15 维 zind 字段
ZIND_FIELDS = [
    "current_ratio", "quick_ratio",
    "assets_turn", "fa_turn",
    "roa", "roe", "npta",
    "profit_to_gr", "op_of_gr", "ebit_of_gr",
    "ocf_to_debt", "q_ocf_to_sales", "ocfps",
    "debt_to_eqt", "assets_to_eqt",
]
assert len(ZIND_FIELDS) == 15

ZIND_MIN_SAMPLES = 10
PREFIX = "fini_"


# ---------------- 工具 ----------------

def setup_logging():
    for h in list(logging.getLogger().handlers):
        logging.getLogger().removeHandler(h)
    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s [%(levelname)s] %(message)s",
        handlers=[
            logging.FileHandler(LOG_PATH, encoding="utf-8"),
            logging.StreamHandler(),
        ],
    )


def load_base():
    """从 derived_financials_v0 提取 (firm_id, year, ts_code) base."""
    log = logging.getLogger()
    log.info(f"读 base: {BASE_PARQUET}")
    if not BASE_PARQUET.exists():
        raise FileNotFoundError(f"base 矩阵不存在: {BASE_PARQUET}")

    df = pd.read_parquet(BASE_PARQUET)
    log.info(f"  shape: {df.shape}")

    # 必须有 firm_id, year, ts_code
    candidates = {
        "firm_id": ["firm_id"],
        "year": ["year"],
        "ts_code": ["ts_code", "stock_code", "code"],
    }
    keys = {}
    for std, opts in candidates.items():
        hit = next((c for c in opts if c in df.columns), None)
        if hit is None:
            raise ValueError(
                f"base 矩阵缺少 {std} 列 (尝试过 {opts}); "
                f"实际列: {list(df.columns)[:30]}"
            )
        keys[std] = hit

    base = df[[keys["firm_id"], keys["year"], keys["ts_code"]]].copy()
    base.columns = ["firm_id", "year", "ts_code"]
    base["year"] = base["year"].astype(int)
    base = base.drop_duplicates()
    log.info(f"  base 唯一 (firm_id, year): {len(base):,}")
    log.info(f"  base 唯一 ts_code: {base['ts_code'].nunique():,}")
    log.info(f"  base 年份: {base['year'].min()}-{base['year'].max()}")
    log.info(f"  base 缺失 ts_code: {base['ts_code'].isna().sum():,}")
    return base


def load_industry():
    """从 stock_basic.db 读 ts_code -> industry 映射."""
    log = logging.getLogger()
    log.info(f"读 stock_basic: {DB_BASIC}")
    if not DB_BASIC.exists():
        raise FileNotFoundError(f"stock_basic.db 不存在: {DB_BASIC}")

    conn = sqlite3.connect(str(DB_BASIC))
    df = pd.read_sql("SELECT ts_code, industry FROM stock_basic", conn)
    conn.close()

    n_no_ind = df["industry"].isna().sum()
    log.info(f"  shape: {df.shape}, industry NaN: {n_no_ind:,}")
    return df.drop_duplicates(subset=["ts_code"])


def load_fina_indicator(base_ts_codes):
    """读 fina_indicator 表, I7 过滤 + I13.3.2 去重 + I10 类型 cast.

    base_ts_codes: 用 base 的 ts_code 做白名单, 缩小 SQL 读取量
    """
    log = logging.getLogger()
    log.info(f"读 fina_indicator: {DB_FIN}")

    needed_cols = ["ts_code", "ann_date", "end_date"] + T0_FIELDS

    # SQL: 仅年报 + 仅 base 中出现的 ts_code
    placeholders = ",".join("?" * len(base_ts_codes))
    cols_str = ", ".join(needed_cols)
    sql = f"""
        SELECT {cols_str}
        FROM fina_indicator
        WHERE end_date LIKE '%1231'
          AND ts_code IN ({placeholders})
    """

    conn = sqlite3.connect(str(DB_FIN))
    t = time.time()
    df = pd.read_sql(sql, conn, params=list(base_ts_codes))
    conn.close()
    log.info(f"  I7 过滤 + ts_code 白名单后: {len(df):,} 行 ({time.time()-t:.1f}s)")

    # I13.3.2: 同 (ts_code, end_date) 取 ann_date 最大者
    df["ann_date"] = pd.to_numeric(df["ann_date"], errors="coerce")
    df = df.sort_values("ann_date", ascending=False)
    n_before = len(df)
    df = df.drop_duplicates(subset=["ts_code", "end_date"], keep="first")
    log.info(f"  I13.3.2 ann_date 最大者去重后: {len(df):,} (减少 {n_before - len(df):,})")

    # I10: 全字段 to_numeric
    coerce_cnt = 0
    for f in T0_FIELDS:
        if f in df.columns:
            before_na = df[f].isna().sum()
            df[f] = pd.to_numeric(df[f], errors="coerce")
            after_na = df[f].isna().sum()
            if after_na > before_na:
                coerce_cnt += 1
    log.info(f"  I10 to_numeric: {coerce_cnt}/{len(T0_FIELDS)} 字段产生新 NaN")

    # 转 year (从 end_date 截前 4 位)
    df["year"] = df["end_date"].str[:4].astype(int)

    yc = df.groupby("year").size()
    log.info(
        f"  年份覆盖: {yc.min()} (year={yc.idxmin()}) - "
        f"{yc.max()} (year={yc.idxmax()})"
    )
    return df


def make_t0(fina_df, base):
    """生成 51 个 t0 列, align 到 base."""
    log = logging.getLogger()
    rename = {f: f"{PREFIX}{f}_t0" for f in T0_FIELDS}
    fina_t0 = fina_df[["ts_code", "year"] + T0_FIELDS].rename(columns=rename)

    out = base.merge(fina_t0, on=["ts_code", "year"], how="left")
    log.info(f"  t0 align 后 shape: {out.shape}")

    # 抽样打印 t0 NaN 率
    nans = []
    for f in T0_FIELDS:
        col = f"{PREFIX}{f}_t0"
        if col in out.columns:
            nans.append((col, out[col].isna().mean()))
    nans.sort(key=lambda x: -x[1])
    log.info("  t0 NaN 率 top 5 高:")
    for col, r in nans[:5]:
        log.info(f"    {col}: {r:.1%}")
    log.info("  t0 NaN 率 top 5 低:")
    for col, r in nans[-5:]:
        log.info(f"    {col}: {r:.1%}")

    return out


def add_t1(out, fina_df):
    """生成 12 个 t1 列: 用 fina_df 自身 year-1 贴到 t 年."""
    log = logging.getLogger()
    fina_t1 = fina_df[["ts_code", "year"] + T1_FIELDS].copy()
    fina_t1["year"] = fina_t1["year"] + 1  # t-1 → t
    rename = {f: f"{PREFIX}{f}_t1" for f in T1_FIELDS}
    fina_t1 = fina_t1.rename(columns=rename)

    out = out.merge(fina_t1, on=["ts_code", "year"], how="left")
    log.info(f"  +t1 shape: {out.shape}")

    # 打印 t1 NaN 率
    log.info("  t1 NaN 率 (高 → 低):")
    nans = []
    for f in T1_FIELDS:
        col = f"{PREFIX}{f}_t1"
        if col in out.columns:
            nans.append((col, out[col].isna().mean()))
    nans.sort(key=lambda x: -x[1])
    for col, r in nans:
        log.info(f"    {col}: {r:.1%}")
    return out


def add_zind(out, industry_df):
    """对 ZIND_FIELDS 做 (year, industry) 行业 z-score, min_samples=10."""
    log = logging.getLogger()
    out = out.merge(industry_df, on="ts_code", how="left")
    n_no_ind = out["industry"].isna().sum()
    log.info(f"  industry NaN: {n_no_ind:,} / {len(out):,} ({n_no_ind/len(out):.1%})")

    for f in ZIND_FIELDS:
        t0_col = f"{PREFIX}{f}_t0"
        z_col = f"{PREFIX}{f}_t0_zind"

        if t0_col not in out.columns:
            log.warning(f"  跳过 {f}: t0 列缺失")
            out[z_col] = np.nan
            continue

        grp = out.groupby(["year", "industry"], dropna=False)[t0_col]
        gmean = grp.transform("mean")
        gstd = grp.transform("std")
        gcnt = grp.transform("count")

        z = (out[t0_col] - gmean) / gstd
        z = z.where((gcnt >= ZIND_MIN_SAMPLES) & (gstd > 0))
        # I13.3.7: industry NaN → zind NaN
        z = z.where(out["industry"].notna())
        out[z_col] = z

        valid = z.notna().sum()
        log.info(f"    {z_col}: valid {valid:,} ({valid/len(out):.1%})")

    return out


def main():
    setup_logging()
    log = logging.getLogger()
    log.info("=" * 70)
    log.info("P4.5.1 Step 2: compute_fina_indicator_v0")
    log.info("=" * 70)
    log.info(f"决策: I7 / I9.1 / I10 / I11 / I12 / I13 / I13.3.1-8")
    log.info(f"字段: {len(T0_FIELDS)} t0 + {len(T1_FIELDS)} t1 + {len(ZIND_FIELDS)} zind")
    log.info("")

    base = load_base()
    industry_df = load_industry()
    fina_df = load_fina_indicator(base["ts_code"].dropna().unique().tolist())

    out = make_t0(fina_df, base)
    out = add_t1(out, fina_df)
    out = add_zind(out, industry_df)

    log.info("")
    log.info(f"最终输出 shape: {out.shape}")
    fini_cols = [c for c in out.columns if c.startswith(PREFIX)]
    log.info(f"  fini_ 列总数: {len(fini_cols)}")
    log.info(f"  含 t0: {sum(1 for c in fini_cols if c.endswith('_t0'))}")
    log.info(f"  含 t1: {sum(1 for c in fini_cols if c.endswith('_t1'))}")
    log.info(f"  含 zind: {sum(1 for c in fini_cols if c.endswith('_zind'))}")

    # 落盘
    out.to_parquet(OUTPUT_PARQUET, index=False)
    log.info(f"[OK] 已写入: {OUTPUT_PARQUET}")
    log.info(f"     {len(out):,} 行 × {len(out.columns)} 列")

    # 完整性 sanity check
    log.info("")
    log.info("[Sanity] 至少有 1 个 fini_ 非 NaN 的行数:")
    has_any = out[fini_cols].notna().any(axis=1).sum()
    log.info(f"  {has_any:,} / {len(out):,} ({has_any/len(out):.1%})")
    if has_any < len(out) * 0.85:
        log.warning(f"  覆盖率偏低 ({has_any/len(out):.1%})！需排查 ts_code 映射或年份对齐")

    log.info("=" * 70)


if __name__ == "__main__":
    main()
